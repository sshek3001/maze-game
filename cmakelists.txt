cmake_minimum_required(VERSION 3.10)
project(MazeSDL2 CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(SDL2 REQUIRED)

add_executable(maze main.cpp)

target_include_directories(maze PRIVATE ${SDL2_INCLUDE_DIRS})
target_link_libraries(maze PRIVATE ${SDL2_LIBRARIES})

if(MSVC)
    target_compile_options(maze PRIVATE /W4)
else()
    target_compile_options(maze PRIVATE -Wall -Wextra)
endif()
